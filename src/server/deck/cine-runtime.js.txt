// Present 3D cinematic presentation engine. Reads deck.json and renders each
// slide as an AI photograph animated with film-like moves and kinetic typography.
// Keys: → / Space next · ← previous · Home/End · F fullscreen · N speaker notes · P auto-play.
import gsap from "gsap";

const deck = await fetch("./deck.json").then((r) => r.json());
// Older decks have no settings; the server validates links, this only guards the engine.
const settings = deck.settings || {};
const webLink = (u, schemes = /^https?:\/\//i) => (typeof u === "string" && schemes.test(u) ? u : "");
const slides = [...deck.slides];
// The owner's Thank You slide closes the deck, with its call-to-action buttons.
const thankYou = settings.thankYou;
if (thankYou?.enabled) {
  slides.push({
    layout: "hero",
    kicker: "",
    title: thankYou.title || "Thank you",
    subtitle: thankYou.message || "",
    bullets: [],
    stats: [],
    notes: "",
    buttons: (thankYou.buttons || []).filter((b) => b && b.label && webLink(b.url, /^(https?:\/\/|mailto:|tel:)/i)),
    // It closes calmly, over the last slide's photograph, in a text style the last slide did not use.
    cine: { text: slides[slides.length - 1]?.cine?.text === "fadeUp" ? "zoom" : "fadeUp", camera: "drift", transition: "glide", pace: "calm", imagePrompt: "", image: slides[slides.length - 1]?.cine?.image || "" },
  });
}
document.title = deck.title;

// ─── Theme ──────────────────────────────────────────────────────────────
const root = document.documentElement;
root.style.setProperty("--bg", deck.theme.background);
root.style.setProperty("--text", deck.theme.text);
root.style.setProperty("--accent", deck.theme.accent);
document.body.dataset.font = deck.theme.font;

// ─── Readability ────────────────────────────────────────────────────────
// Text must always read clearly over its photograph. Each image is measured
// where the slide's text sits: over a bright area the text turns dark on a light
// shade, over a dark area it is light on a dark shade (slide data-backdrop).
const hexLum = (hex) => {
  const [r, g, b] = [1, 3, 5].map((i) => parseInt(String(hex).slice(i, i + 2), 16) / 255);
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
};
const textIsLight = hexLum(deck.theme.text) > 0.5;
root.style.setProperty("--ink-light", textIsLight ? deck.theme.text : "#f6f3ee");
root.style.setProperty("--ink-dark", textIsLight ? "#15120f" : deck.theme.text);
const THEME_BACKDROP = textIsLight ? "dark" : "light";
// The part of the picture behind the text, per layout: [x0, y0, x1, y1] as fractions.
const TEXT_AREA = { left: [0, 0.2, 0.5, 0.85], right: [0.5, 0.2, 1, 0.85], hero: [0.15, 0.25, 0.85, 0.75], center: [0.15, 0.25, 0.85, 0.75], stats: [0.12, 0.4, 0.88, 0.95] };
const backdrops = new Map(); // slide index → "light" | "dark"

/** Mean brightness (0-1) of the image where the text sits, or null if it cannot be read (e.g. another site's image). */
function measure(img, layout) {
  try {
    const W = 64, H = 36;
    const c = document.createElement("canvas");
    c.width = W;
    c.height = H;
    const g = c.getContext("2d", { willReadFrequently: true });
    g.drawImage(img, 0, 0, W, H);
    const [x0, y0, x1, y1] = TEXT_AREA[layout] ?? TEXT_AREA.center;
    const d = g.getImageData(Math.floor(x0 * W), Math.floor(y0 * H), Math.max(1, Math.round((x1 - x0) * W)), Math.max(1, Math.round((y1 - y0) * H))).data;
    let sum = 0;
    for (let i = 0; i < d.length; i += 4) sum += (0.2126 * d[i] + 0.7152 * d[i + 1] + 0.0722 * d[i + 2]) / 255;
    return sum / (d.length / 4);
  } catch {
    return null;
  }
}
const backdropFor = (brightness) => (brightness === null ? THEME_BACKDROP : brightness > 0.58 ? "light" : brightness < 0.42 ? "dark" : THEME_BACKDROP);
// Every image is loaded and measured up front, so a slide's colours are right from its first frame.
slides.forEach((s, i) => {
  if (!s.cine?.image) return;
  const img = new Image();
  if (s.cine.image.startsWith("/")) img.crossOrigin = "anonymous";
  img.addEventListener("load", () => backdrops.set(i, backdropFor(measure(img, s.layout))));
  img.src = s.cine.image;
});

// ─── Kinetic typography ─────────────────────────────────────────────────
// Every slide's text animates in with its own style (slide.cine.text).
// Text is split into words/characters for the animation only; the element keeps
// its full text as an aria-label so screen readers read it normally.
// Animations run on real time even when frames are slow: GSAP's default lag
// smoothing would otherwise slow them down until text appears stuck.
gsap.ticker.lagSmoothing(0);
const REDUCED_MOTION = window.matchMedia?.("(prefers-reduced-motion: reduce)").matches ?? false;
const PACE = { calm: 1.3, normal: 1, energetic: 0.72 };
const motionOf = (slide) => slide.cine ?? { text: "fadeUp", camera: "static", transition: "glide", pace: "normal" };
const paceOf = (slide) => PACE[motionOf(slide).pace] ?? 1;

function splitText(node, mode) {
  const text = node.textContent;
  node.setAttribute("aria-label", text);
  node.textContent = "";
  const words = text.split(/(\s+)/);
  const parts = [];
  for (const w of words) {
    if (!w) continue;
    if (/^\s+$/.test(w)) {
      node.append(document.createTextNode(" "));
      continue;
    }
    const word = el("span", "kw");
    word.setAttribute("aria-hidden", "true");
    if (mode === "chars") {
      for (const c of w) {
        const ch = el("span", "kc", c);
        word.append(ch);
        parts.push(ch);
      }
    } else parts.push(word), (word.textContent = w);
    node.append(word);
  }
  return parts;
}

const rand = (a, b) => a + Math.random() * (b - a);

/** Animate a freshly built slide's text. Returns the GSAP timeline. */
function animateText(box, slide, delay, dir) {
  const style = REDUCED_MOTION ? "fadeUp" : motionOf(slide).text;
  const k = paceOf(slide);
  const title = box.querySelector(".title");
  const rest = [...box.children].filter((c) => c !== title);
  const tl = gsap.timeline({ delay });
  box.dataset.style = style;
  gsap.set(box.children, { opacity: 1 });

  if (style === "glitch") {
    const chars = splitText(title, "chars");
    gsap.set(chars, { opacity: 0 });
    title.classList.add("glitching");
    tl.fromTo(chars, { opacity: 0, x: () => rand(-18, 18), y: () => rand(-6, 6), skewX: () => rand(-30, 30) }, { opacity: 1, x: 0, y: 0, skewX: 0, duration: 0.09 * k, ease: "steps(3)", stagger: { each: 0.022 * k, from: "random" } });
    tl.call(() => title.classList.remove("glitching"), [], "+=0.25");
    tl.fromTo(rest, { opacity: 0, x: () => rand(-10, 10), filter: "blur(4px)" }, { opacity: 1, x: 0, filter: "blur(0px)", duration: 0.35 * k, ease: "power2.out", stagger: 0.06 * k }, 0.35 * k);
  } else if (style === "zoom") {
    tl.fromTo(title, { opacity: 0, scale: 1.9, filter: "blur(18px)", letterSpacing: "0.25em" }, { opacity: 1, scale: 1, filter: "blur(0px)", letterSpacing: "", duration: 1.15 * k, ease: "expo.out", clearProps: "letterSpacing,filter" });
    tl.fromTo(rest, { opacity: 0, scale: 0.92 }, { opacity: 1, scale: 1, duration: 0.7 * k, ease: "power3.out", stagger: 0.08 * k }, 0.45 * k);
  } else if (style === "split") {
    const words = splitText(title, "words");
    tl.fromTo(words, { opacity: 0, x: (i) => (i % 2 ? 1 : -1) * 110, filter: "blur(6px)" }, { opacity: 1, x: 0, filter: "blur(0px)", duration: 0.9 * k, ease: "power4.out", stagger: 0.05 * k });
    tl.fromTo(rest, { opacity: 0, x: (i) => (i % 2 ? 1 : -1) * 50 }, { opacity: 1, x: 0, duration: 0.7 * k, ease: "power3.out", stagger: 0.07 * k }, 0.3 * k);
  } else if (style === "rotate3d") {
    const words = splitText(title, "words");
    tl.fromTo(words, { opacity: 0, rotateX: -100, y: 24, transformOrigin: "50% 100% -30px" }, { opacity: 1, rotateX: 0, y: 0, duration: 0.95 * k, ease: "back.out(1.4)", stagger: 0.07 * k });
    tl.fromTo(rest, { opacity: 0, rotateY: 35 * dir, transformOrigin: dir > 0 ? "0% 50%" : "100% 50%" }, { opacity: 1, rotateY: 0, duration: 0.8 * k, ease: "power3.out", stagger: 0.08 * k }, 0.35 * k);
  } else if (style === "wipe") {
    // An accent bar sweeps across; the title is revealed behind it as it retracts.
    const inner = el("span", "wt", title.textContent);
    const bar = el("span", "wipe-bar");
    bar.setAttribute("aria-hidden", "true");
    title.textContent = "";
    title.append(inner, bar);
    gsap.set(inner, { clipPath: "inset(0 100% 0 0)" });
    tl.fromTo(bar, { scaleX: 0, transformOrigin: "0% 50%" }, { scaleX: 1, duration: 0.45 * k, ease: "power3.inOut" });
    tl.set(inner, { clipPath: "inset(0 0% 0 0)" });
    tl.to(bar, { scaleX: 0, transformOrigin: "100% 50%", duration: 0.45 * k, ease: "power3.inOut" });
    tl.call(() => bar.remove());
    tl.set(inner, { clearProps: "clipPath" });
    tl.fromTo(rest, { opacity: 0, clipPath: "inset(0 100% 0 0)" }, { opacity: 1, clipPath: "inset(0 0% 0 0)", duration: 0.6 * k, ease: "power3.out", stagger: 0.08 * k, clearProps: "clipPath" }, 0.5 * k);
  } else if (style === "letters") {
    const chars = splitText(title, "chars");
    tl.fromTo(chars, { opacity: 0, y: 46, rotate: 10 }, { opacity: 1, y: 0, rotate: 0, duration: 0.7 * k, ease: "power4.out", stagger: Math.min(0.045, 1.1 / chars.length) * k });
    tl.fromTo(rest, { opacity: 0, y: 18 }, { opacity: 1, y: 0, duration: 0.6 * k, ease: "power3.out", stagger: 0.07 * k }, "-=0.35");
  } else {
    // fadeUp: smooth slide-in in the travel direction
    tl.fromTo(box.children, { opacity: 0, y: 34, x: 30 * dir }, { opacity: 1, y: 0, x: 0, duration: 0.85 * k, ease: "power3.out", stagger: 0.08 * k });
  }
  // Bullets, stats and buttons cascade in after their block.
  const items = box.querySelectorAll(".bullets li, .stat, .cta-btn");
  if (items.length && style !== "fadeUp") tl.fromTo(items, { opacity: 0, y: 14 }, { opacity: 1, y: 0, duration: 0.5 * k, ease: "power2.out", stagger: 0.07 * k }, "-=0.3");
  return tl;
}

/** The leaving text exits in a way that matches how it came in. */
function exitText(page, dir) {
  const style = page.querySelector(".content")?.dataset.style;
  const out = { opacity: 0, duration: TEXT_OUT, ease: "power2.in", onComplete: () => page.remove() };
  if (style === "zoom") gsap.to(page, { ...out, scale: 1.15, filter: "blur(10px)" });
  else if (style === "glitch") gsap.to(page, { ...out, x: -40 * dir, skewX: 20, ease: "steps(4)" });
  else if (style === "rotate3d") gsap.to(page, { ...out, rotateY: -25 * dir, transformOrigin: dir > 0 ? "0% 50%" : "100% 50%" });
  else gsap.to(page, { ...out, x: -70 * dir });
}

// ─── HUD and slides ─────────────────────────────────────────────────────
const slideEl = document.getElementById("slide");
const dotsEl = document.getElementById("dots");
const counterEl = document.getElementById("counter");
const chipTitleEl = document.getElementById("chip-title");
const progressEl = document.getElementById("progress");
const notesEl = document.getElementById("notes");
const hintEl = document.getElementById("hint");
const pad = (n) => String(n).padStart(2, "0");

function el(tag, cls, text) {
  const e = document.createElement(tag);
  if (cls) e.className = cls;
  if (text !== undefined) e.textContent = text; // text only: deck content is never parsed as HTML
  return e;
}

let textTween = null;
const TEXT_OUT = 0.32;
// Each slide lives in its own page element. On a change, every page on screen
// leaves (in the travel direction) before the new one comes in, so two slides
// are never visible together, even when navigating quickly.
function renderSlide(slide, delay, dir = 1) {
  textTween?.kill();
  for (const old of [...slideEl.children]) {
    gsap.killTweensOf([old, ...old.querySelectorAll("*")]);
    if (old.dataset.leaving) {
      old.remove(); // already on its way out: drop it now
      continue;
    }
    old.dataset.leaving = "1";
    old.setAttribute("aria-hidden", "true");
    exitText(old, dir);
  }
  const page = el("div", `slide layout-${slide.layout}`);
  const at = slides.indexOf(slide);
  page.dataset.backdrop = backdrops.get(at) ?? THEME_BACKDROP;
  if (slide.cine?.image) {
    const art = el("img", "slide-art");
    if (slide.cine.image.startsWith("/")) art.crossOrigin = "anonymous";
    // Not measured yet (still loading): measured as soon as it arrives.
    if (!backdrops.has(at)) {
      art.addEventListener("load", () => {
        backdrops.set(at, backdropFor(measure(art, slide.layout)));
        page.dataset.backdrop = backdrops.get(at);
      });
    }
    art.src = slide.cine.image;
    art.alt = "";
    art.setAttribute("aria-hidden", "true");
    page.append(art, el("div", "slide-shade"));
    const pace = paceOf(slide);
    const transition = REDUCED_MOTION ? "glide" : motionOf(slide).transition;
    const cameraMove = REDUCED_MOTION ? "static" : motionOf(slide).camera;
    // The image's slow move while the slide is on screen (slide.cine.camera).
    const imageMoves = {
      push: [{ scale: 1.025 }, { scale: 1.085 }],
      pull: [{ scale: 1.09 }, { scale: 1.025 }],
      orbit: [{ scale: 1.06, x: -14 }, { scale: 1.06, x: 14 }],
      crane: [{ scale: 1.055, y: 10 }, { scale: 1.055, y: -10 }],
      drift: [{ scale: 1.055, x: -16 }, { scale: 1.055, x: 16 }],
      static: [{ scale: 1.025 }, { scale: 1.025 }],
    };
    const from = transition === "whip" ? { opacity: 0, x: -dir * 90, scale: 1.12, filter: "blur(8px)" }
      : transition === "rise" ? { opacity: 0, y: 70, scale: 1.06 }
        : transition === "spiral" ? { opacity: 0, scale: 1.2, rotate: dir * 2, filter: "blur(10px)" }
          : transition === "zoom" ? { opacity: 0, scale: 1.22, filter: "blur(12px)" }
            : { opacity: 0, x: dir * 42, scale: 1.1 };
    gsap.fromTo(art, from, {
      opacity: 1, x: 0, y: 0, scale: 1.02, rotate: 0, filter: "blur(0px)",
      duration: (transition === "whip" ? 0.72 : 1.15) * pace,
      ease: transition === "whip" ? "power4.out" : "power3.out",
      delay: Math.max(0, delay * 0.45),
      onComplete: () => {
        if (!REDUCED_MOTION && page.isConnected) {
          const [from, to] = imageMoves[cameraMove] ?? imageMoves.static;
          gsap.fromTo(art, from, { ...to, duration: 10 * pace, ease: "sine.inOut", yoyo: true, repeat: -1 });
        }
      },
    });
    gsap.fromTo(page.querySelector(".slide-shade"), { opacity: 0.45 }, { opacity: 1, duration: 0.8 * pace, delay: Math.max(0, delay * 0.45) });
  }
  const box = el("div", "content");
  if (slide.kicker) box.append(el("p", "kicker", slide.kicker));
  box.append(el(slide.layout === "hero" ? "h1" : "h2", "title", slide.title));
  if (slide.subtitle) box.append(el("p", "subtitle", slide.subtitle));
  if (slide.bullets.length) {
    const ul = el("ul", "bullets");
    for (const b of slide.bullets) ul.append(el("li", "", b));
    box.append(ul);
  }
  if (slide.layout === "stats" && slide.stats.length) {
    const grid = el("div", "stats");
    grid.style.setProperty("--n", String(Math.min(4, slide.stats.length)));
    for (const s of slide.stats) {
      const card = el("div", "stat");
      card.append(el("strong", "", s.value), el("span", "", s.label));
      grid.append(card);
    }
    box.append(grid);
  }
  if (slide.buttons?.length) {
    const row = el("div", "cta");
    slide.buttons.forEach((b, i) => {
      const a = el("a", i === 0 ? "cta-btn primary" : "cta-btn", b.label);
      a.href = b.url;
      a.target = "_blank";
      a.rel = "noopener noreferrer";
      a.addEventListener("click", (e) => e.stopPropagation());
      row.append(a);
    });
    box.append(row);
  }
  page.append(box);
  slideEl.append(page);
  notesEl.textContent = slide.notes || "No speaker notes for this slide.";
  // The new text starts only after the old text has left, in this slide's own style.
  gsap.set(box.children, { opacity: 0 });
  textTween = animateText(box, slide, Math.max(delay, TEXT_OUT + 0.05), dir);
}

slides.forEach((_, i) => {
  const d = el("button", "dot");
  d.setAttribute("aria-label", `Go to slide ${i + 1}`);
  d.addEventListener("click", (e) => {
    e.stopPropagation();
    go(i);
  });
  dotsEl.append(d);
});

// ─── Travel ─────────────────────────────────────────────────────────────
let index = -1;
let lastNav = 0;

function go(next, instant = false) {
  next = Math.max(0, Math.min(slides.length - 1, next));
  if (next === index) return;
  // Allow quick consecutive presses: a new target simply re-aims the travel.
  const now = performance.now();
  if (now - lastNav < 250 && !instant) return;
  lastNav = now;
  const distance = index < 0 ? 0 : Math.abs(next - index);
  const dir = index < 0 || next > index ? 1 : -1;
  index = next;
  const slide = slides[next];
  if (instant || distance === 0) renderSlide(slide, 0.1, dir);
  else {
    // The transition's length paces the hand-over (slide.cine.transition shapes the image's entry).
    const kind = REDUCED_MOTION ? "glide" : motionOf(slide).transition;
    const k = paceOf(slide);
    const duration = (kind === "whip" ? 0.95 : Math.min(2.6, 1.6 + 0.25 * (distance - 1))) * k;
    renderSlide(slide, duration * 0.5, dir);
  }
  [...dotsEl.children].forEach((d, i) => d.classList.toggle("active", i <= index));
  counterEl.textContent = `${pad(index + 1)} / ${pad(slides.length)}`;
  chipTitleEl.textContent = slide.title;
  progressEl.style.transform = `scaleX(${(index + 1) / slides.length})`;
  history.replaceState(null, "", `#${index + 1}`);
  // Tell the hosting page which slide is shown (the editor reopens a rebuilt preview on it).
  try {
    if (window.parent !== window) window.parent.postMessage({ source: "deck3d-preview", type: "slide", index: index + 1 }, "*");
  } catch {
    /* no parent */
  }
  schedulePlay();
  scheduleThumbnail();
}

const next = () => go(index + 1);
const prev = () => go(index - 1);
document.getElementById("next").addEventListener("click", (e) => { e.stopPropagation(); next(); });
document.getElementById("prev").addEventListener("click", (e) => { e.stopPropagation(); prev(); });
document.getElementById("fs").addEventListener("click", (e) => { e.stopPropagation(); toggleFullscreen(); });

function toggleFullscreen() {
  if (document.fullscreenElement) document.exitFullscreen?.();
  else document.documentElement.requestFullscreen?.().catch(() => {});
}

window.addEventListener("keydown", (e) => {
  if (["ArrowRight", "ArrowDown", "PageDown", " ", "Enter"].includes(e.key)) { e.preventDefault(); next(); }
  else if (["ArrowLeft", "ArrowUp", "PageUp", "Backspace"].includes(e.key)) { e.preventDefault(); prev(); }
  else if (e.key === "Home") go(0);
  else if (e.key === "End") go(slides.length - 1);
  else if (e.key.toLowerCase() === "f") toggleFullscreen();
  else if (e.key.toLowerCase() === "n") document.body.classList.toggle("show-notes");
  else if (e.key.toLowerCase() === "p") { if (!playing && index >= slides.length - 1) go(0); setPlaying(!playing); }
});
document.getElementById("stage").addEventListener("click", (e) => {
  if (e.target.closest("button, a")) return;
  (e.clientX < window.innerWidth * 0.25 ? prev : next)();
});
let touchX = null;
window.addEventListener("touchstart", (e) => (touchX = e.touches[0].clientX), { passive: true });
window.addEventListener("touchend", (e) => {
  if (touchX === null) return;
  const dx = e.changedTouches[0].clientX - touchX;
  if (Math.abs(dx) > 50) (dx < 0 ? next : prev)();
  touchX = null;
});

// Mouse wheel / trackpad: one slide per gesture (accumulate, then lock until the
// gesture's events stop, so trackpad inertia does not skip several slides).
let wheelAcc = 0;
let wheelLockUntil = 0;
window.addEventListener(
  "wheel",
  (e) => {
    e.preventDefault();
    const now = performance.now();
    if (now < wheelLockUntil) {
      wheelLockUntil = Math.max(wheelLockUntil, now + 300);
      return;
    }
    wheelAcc += Math.abs(e.deltaY) >= Math.abs(e.deltaX) ? e.deltaY : e.deltaX;
    if (Math.abs(wheelAcc) >= 40) {
      (wheelAcc > 0 ? next : prev)();
      wheelAcc = 0;
      wheelLockUntil = now + 700;
    }
  },
  { passive: false },
);

// Navigation commands from the page hosting this deck (it forwards keys when the
// frame does not have focus). Only the direct parent is accepted, and only these actions.
window.addEventListener("message", (e) => {
  if (e.source !== window.parent || !e.data || e.data.type !== "deck3d-deck") return;
  const action = e.data.action;
  if (action === "next") next();
  else if (action === "prev") prev();
  else if (action === "first") go(0);
  else if (action === "last") go(slides.length - 1);
  else if (action === "thumbnail") {
    thumbWanted = true;
    scheduleThumbnail();
  }
  // The PHP viewer narrates the deck and moves it on when a slide's voice ends;
  // the deck's own timer must not move it as well.
  else if (action === "autoplay") setPlaying(Boolean(e.data.on));
});

window.addEventListener("hashchange", () => {
  const n = parseInt(location.hash.slice(1), 10);
  if (Number.isFinite(n)) go(n - 1);
});

// ─── Thumbnail ──────────────────────────────────────────────────────────
// The editor asks for a picture of slide 1 (the presentation listing shows it).
// It is drawn once slide 1 is on screen and its text has settled: the slide's
// picture, then its text where it sits, in its own fonts and colours.
let thumbWanted = false;
let thumbTimer = null;
function scheduleThumbnail() {
  clearTimeout(thumbTimer);
  if (!thumbWanted || index !== 0) return;
  thumbTimer = setTimeout(() => {
    if (!thumbWanted || index !== 0) return;
    thumbWanted = false;
    try {
      window.parent.postMessage({ source: "deck3d-preview", type: "thumbnail", image: captureThumbnail() }, "*");
    } catch {
      /* no parent, or the picture could not be read */
    }
  }, 3200);
}

function captureThumbnail() {
  const W = 640, H = 360;
  const vw = window.innerWidth, vh = window.innerHeight;
  const c = document.createElement("canvas");
  c.width = W;
  c.height = H;
  const g = c.getContext("2d");
  // The screen is scaled to cover the thumbnail, like a cover-fitted picture.
  const s = Math.max(W / vw, H / vh);
  const map = { s, x: (v) => (v - vw / 2) * s + W / 2, y: (v) => (v - vh / 2) * s + H / 2, W, H, vw, vh };
  g.fillStyle = deck.theme.background;
  g.fillRect(0, 0, W, H);
  drawThumbBackground(g, map);
  const page = [...slideEl.children].filter((p) => !p.dataset.leaving).pop();
  for (const node of page ? page.querySelectorAll(".kicker, .title, .subtitle, .bullets li") : []) {
    const r = node.getBoundingClientRect();
    if (!r.width || !r.height) continue;
    const cs = getComputedStyle(node);
    let text = (node.getAttribute("aria-label") || node.textContent || "").replace(/\s+/g, " ").trim();
    if (!text) continue;
    if (cs.textTransform === "uppercase") text = text.toUpperCase();
    const size = parseFloat(cs.fontSize) * s;
    g.font = `${cs.fontStyle} ${cs.fontWeight} ${size}px ${cs.fontFamily}`;
    if ("letterSpacing" in g) g.letterSpacing = `${(parseFloat(cs.letterSpacing) || 0) * s}px`;
    g.fillStyle = cs.color;
    g.globalAlpha = Math.max(0.6, parseFloat(cs.opacity) || 1);
    const [cr, cg, cb] = (cs.color.match(/\d+(\.\d+)?/g) || [255, 255, 255]).map(Number);
    const lightText = 0.2126 * cr + 0.7152 * cg + 0.0722 * cb > 140;
    g.shadowColor = lightText ? "rgba(0,0,0,0.55)" : "rgba(255,255,255,0.6)";
    g.shadowBlur = 10 * s;
    g.textBaseline = "top";
    const pad = node.matches(".bullets li") ? (parseFloat(cs.paddingLeft) || 0) * s : 0;
    const left = map.x(r.left) + pad;
    const maxW = r.width * s - pad;
    const lineH = (parseFloat(cs.lineHeight) || parseFloat(cs.fontSize) * 1.2) * s;
    const lines = [];
    let line = "";
    for (const word of text.split(" ")) {
      const t = line ? `${line} ${word}` : word;
      if (line && g.measureText(t).width > maxW) {
        lines.push(line);
        line = word;
      } else line = t;
    }
    if (line) lines.push(line);
    const align = cs.textAlign === "center" ? "center" : cs.textAlign === "right" || cs.textAlign === "end" ? "right" : "left";
    g.textAlign = align;
    const x = align === "center" ? left + maxW / 2 : align === "right" ? left + maxW : left;
    const top = map.y(r.top) + Math.max(0, (r.height * s - lines.length * lineH) / 2);
    lines.forEach((l, i) => g.fillText(l, x, top + i * lineH));
    if (pad) {
      g.fillStyle = getComputedStyle(document.documentElement).getPropertyValue("--accent") || deck.theme.accent;
      g.fillRect(map.x(r.left), top + lineH * 0.55, 12 * s, Math.max(1, 2 * s));
    }
  }
  g.globalAlpha = 1;
  return c.toDataURL("image/jpeg", 0.82);
}

function drawThumbBackground(g, map) {
  const page = [...slideEl.children].filter((p) => !p.dataset.leaving).pop();
  const art = page?.querySelector(".slide-art");
  if (art && art.complete && art.naturalWidth) {
    const k = Math.max(map.vw / art.naturalWidth, map.vh / art.naturalHeight) * map.s;
    const dw = art.naturalWidth * k, dh = art.naturalHeight * k;
    g.drawImage(art, (map.W - dw) / 2, (map.H - dh) / 2, dw, dh);
  }
  const shade = page?.dataset.backdrop === "light" ? "250,248,244" : "5,6,10";
  const layout = (page?.className.match(/layout-(\w+)/) || [])[1];
  const side = layout === "left" || layout === "right";
  const grad = side ? (layout === "left" ? g.createLinearGradient(0, 0, map.W, 0) : g.createLinearGradient(map.W, 0, 0, 0)) : g.createLinearGradient(0, map.H, 0, 0);
  const stops = side ? [[0, 0.9], [0.42, 0.62], [1, 0.14]] : [[0, 0.76], [0.54, 0.42], [1, 0.55]];
  for (const [at, a] of stops) grad.addColorStop(at, `rgba(${shade},${a})`);
  g.fillStyle = grad;
  g.fillRect(0, 0, map.W, map.H);
}

// ─── Auto-play ──────────────────────────────────────────────────────────
// Moves on by itself every few seconds (no narration) and stops on the last slide.
const playBtn = document.getElementById("play");
const autoplaySeconds = Math.min(60, Math.max(3, Number(settings.autoplay?.seconds) || 7));
let playing = false;
let playTimer = null;

function schedulePlay() {
  clearTimeout(playTimer);
  playTimer = null;
  if (!playing) return;
  if (index >= slides.length - 1) {
    setPlaying(false);
    return;
  }
  playTimer = setTimeout(next, autoplaySeconds * 1000);
}

function setPlaying(on) {
  playing = on;
  playBtn.classList.toggle("playing", on);
  playBtn.setAttribute("aria-label", on ? "Pause auto-play (P)" : "Auto-play (P)");
  playBtn.title = on ? "Pause (P)" : "Auto-play (P)";
  playBtn.innerHTML = on ? "&#10074;&#10074;" : "&#9654;";
  schedulePlay();
}
playBtn.addEventListener("click", (e) => {
  e.stopPropagation();
  // Playing from the last slide starts again from the first.
  if (!playing && index >= slides.length - 1) go(0);
  setPlaying(!playing);
});

    wrap.href = link;
    wrap.target = "_blank";
    wrap.rel = "noopener noreferrer";
  }
  if (brandLogo) {

    img.src = brandLogo;
    img.alt = footer.name || "";
    img.addEventListener("error", () => img.remove());
    wrap.append(img);
  }
  if (footer.name) wrap.append(el("span", "", footer.name));
  brandEl.append(wrap);
  brandEl.hidden = false;
}
